Attribute VB_Name = "modV1CellFieldRelation"
Option Explicit

Public Const V1_CELL_FIELD_RELATION_SHEET As String = "V1_Cell_Field_Relation"
Public Const V1_SINGLE_RUN_NAME As String = "Single Run"

' Set to True if you want:
' DESCRIPTION = <description text> & " at " & <run name>
Public Const V1_APPEND_RUN_TO_DESCRIPTION As Boolean = False

Private Type CellFieldBlock
    iobRow As Long
    fieldRow As Long
    labelCol As Long
    explicitRunName As String
End Type

Public Sub CreateWorkbookCellFieldRelation()

    Dim wb As Workbook
    Dim ws As Worksheet
    Dim reportWs As Worksheet
    Dim blocks() As CellFieldBlock
    Dim blockCount As Long
    Dim relationCount As Long
    Dim outRow As Long

    Dim oldScreenUpdating As Boolean
    Dim oldEnableEvents As Boolean
    Dim oldDisplayAlerts As Boolean
    Dim oldStatusBar As Variant

    On Error GoTo Fail

    Set wb = ActiveWorkbook

    If wb Is Nothing Then
        Err.Raise vbObjectError + 5000, , "No active workbook found."
    End If

    oldScreenUpdating = Application.ScreenUpdating
    oldEnableEvents = Application.EnableEvents
    oldDisplayAlerts = Application.DisplayAlerts
    oldStatusBar = Application.StatusBar

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.StatusBar = "Creating workbook-level V1 Cell-Field Relation..."

    DeleteWorksheetIfExists wb, V1_CELL_FIELD_RELATION_SHEET

    Set reportWs = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    reportWs.Name = V1_CELL_FIELD_RELATION_SHEET

    WriteCellFieldRelationHeaders reportWs
    reportWs.Columns("A:F").NumberFormat = "@"

    outRow = 2

    For Each ws In wb.Worksheets

        If Not IsGeneratedV1ReportSheet(ws.Name) Then

            Application.StatusBar = "Scanning sheet: " & ws.Name

            blockCount = DetectCellFieldBlocks(ws, blocks)

            If blockCount > 0 Then
                relationCount = relationCount + AppendBlocksToReport( _
                    ws:=ws, _
                    reportWs:=reportWs, _
                    blocks:=blocks, _
                    blockCount:=blockCount, _
                    outRow:=outRow)
            End If

        End If

    Next ws

    With reportWs
        .Rows(1).Font.Bold = True
        .Columns("A:F").AutoFit
        .Range("A1:F1").AutoFilter
    End With

CleanExit:
    Application.StatusBar = oldStatusBar
    Application.DisplayAlerts = oldDisplayAlerts
    Application.EnableEvents = oldEnableEvents
    Application.ScreenUpdating = oldScreenUpdating

    Debug.Print "Workbook-level V1 Cell-Field Relation complete."
    Debug.Print "Workbook: " & wb.Name
    Debug.Print "Relations created: " & relationCount
    Debug.Print "Report sheet: " & V1_CELL_FIELD_RELATION_SHEET

    Exit Sub

Fail:
    Application.StatusBar = oldStatusBar
    Application.DisplayAlerts = oldDisplayAlerts
    Application.EnableEvents = oldEnableEvents
    Application.ScreenUpdating = oldScreenUpdating

    Err.Raise Err.Number, "CreateWorkbookCellFieldRelation", Err.Description

End Sub

Private Function DetectCellFieldBlocks( _
    ByVal ws As Worksheet, _
    ByRef blocks() As CellFieldBlock) As Long

    Dim usedRng As Range
    Dim data As Variant
    Dim r As Long
    Dim labelCol As Long
    Dim n As Long
    Dim iobLabel As String
    Dim fieldLabel As String

    Set usedRng = ws.UsedRange

    If usedRng.Rows.Count < 2 Or usedRng.Columns.Count < 2 Then
        DetectCellFieldBlocks = 0
        Exit Function
    End If

    data = usedRng.Value2

    ReDim blocks(1 To 1)
    n = 0

    For r = 1 To UBound(data, 1) - 1

        For labelCol = 1 To UBound(data, 2)

            iobLabel = CellText(data(r, labelCol))
            fieldLabel = CellText(data(r + 1, labelCol))

            If IsIOBLabel(iobLabel) And IsFieldNameLabel(fieldLabel) Then

                n = n + 1

                If n > UBound(blocks) Then
                    ReDim Preserve blocks(1 To n * 2)
                End If

                blocks(n).iobRow = usedRng.Row + r - 1
                blocks(n).fieldRow = usedRng.Row + r
                blocks(n).labelCol = usedRng.Column + labelCol - 1
                blocks(n).explicitRunName = ExtractExplicitRunName(iobLabel, fieldLabel)

            End If

        Next labelCol

    Next r

    If n > 0 Then
        ReDim Preserve blocks(1 To n)
    End If

    DetectCellFieldBlocks = n

End Function

Private Function AppendBlocksToReport( _
    ByVal ws As Worksheet, _
    ByVal reportWs As Worksheet, _
    ByRef blocks() As CellFieldBlock, _
    ByVal blockCount As Long, _
    ByRef outRow As Long) As Long

    Dim i As Long
    Dim c As Long
    Dim lastDataCol As Long
    Dim runName As String
    Dim iobValue As String
    Dim fieldName As String
    Dim descriptionBase As String
    Dim descriptionText As String
    Dim cellAddress As String
    Dim relationsAdded As Long
    Dim populatedBlockCount As Long
    Dim descriptionRow As Long

    populatedBlockCount = CountPopulatedBlocks(ws, blocks, blockCount)

    If populatedBlockCount = 0 Then
        AppendBlocksToReport = 0
        Exit Function
    End If

    descriptionRow = FindSharedDescriptionRow(ws, blocks, blockCount)

    Debug.Print "Sheet: " & ws.Name
    Debug.Print "Detected blocks: " & blockCount
    Debug.Print "Populated blocks: " & populatedBlockCount
    Debug.Print "Description row: " & descriptionRow

    For i = 1 To blockCount

        If Not BlockHasRelationPairs(ws, blocks(i)) Then
            GoTo NextBlock
        End If

        runName = ResolveRunNameOrBlank( _
            explicitRunName:=blocks(i).explicitRunName, _
            populatedBlockCountOnSheet:=populatedBlockCount)

        If Len(runName) = 0 Then
            Debug.Print "Skipped populated block with missing run name."
            Debug.Print "Sheet: " & ws.Name
            Debug.Print "Block index: " & i
            Debug.Print "I/O/B row: " & blocks(i).iobRow
            Debug.Print "Field row: " & blocks(i).fieldRow
            GoTo NextBlock
        End If

        lastDataCol = LastUsedColumnForBlock( _
            ws:=ws, _
            iobRow:=blocks(i).iobRow, _
            fieldRow:=blocks(i).fieldRow, _
            descriptionRow:=descriptionRow)

        For c = blocks(i).labelCol + 1 To lastDataCol

            iobValue = CellText(ws.cells(blocks(i).iobRow, c).Value2)
            fieldName = CellText(ws.cells(blocks(i).fieldRow, c).Value2)

            If Len(iobValue) > 0 And Len(fieldName) > 0 Then

                If descriptionRow > 0 Then
                    descriptionBase = CellText(ws.cells(descriptionRow, c).Value2)
                Else
                    descriptionBase = vbNullString
                End If

                descriptionText = BuildDescription(descriptionBase, fieldName, runName)
                cellAddress = Row2Address(ws, c)

                reportWs.cells(outRow, 1).value = CleanText(ws.Name)
                reportWs.cells(outRow, 2).value = CleanText(cellAddress)
                reportWs.cells(outRow, 3).value = CleanText(runName)
                reportWs.cells(outRow, 4).value = CleanText(fieldName)
                reportWs.cells(outRow, 5).value = CleanText(iobValue)
                reportWs.cells(outRow, 6).value = CleanText(descriptionText)

                outRow = outRow + 1
                relationsAdded = relationsAdded + 1

            End If

        Next c

NextBlock:
    Next i

    AppendBlocksToReport = relationsAdded

End Function

Private Function FindSharedDescriptionRow( _
    ByVal ws As Worksheet, _
    ByRef blocks() As CellFieldBlock, _
    ByVal blockCount As Long) As Long

    Dim usedRng As Range
    Dim maxFieldRow As Long
    Dim r As Long
    Dim c As Long
    Dim i As Long
    Dim firstUsedRow As Long
    Dim lastUsedRow As Long
    Dim firstUsedCol As Long
    Dim lastUsedCol As Long
    Dim labelText As String

    Set usedRng = ws.UsedRange

    firstUsedRow = usedRng.Row
    lastUsedRow = usedRng.Row + usedRng.Rows.Count - 1
    firstUsedCol = usedRng.Column
    lastUsedCol = usedRng.Column + usedRng.Columns.Count - 1

    maxFieldRow = 0

    For i = 1 To blockCount
        If blocks(i).fieldRow > maxFieldRow Then
            maxFieldRow = blocks(i).fieldRow
        End If
    Next i

    For r = maxFieldRow + 1 To lastUsedRow

        For c = firstUsedCol To lastUsedCol

            labelText = CellText(ws.cells(r, c).Value2)

            If IsDescriptionLabel(labelText) Then
                FindSharedDescriptionRow = r
                Exit Function
            End If

        Next c

    Next r

    For r = firstUsedRow To lastUsedRow

        For c = firstUsedCol To lastUsedCol

            labelText = CellText(ws.cells(r, c).Value2)

            If IsDescriptionLabel(labelText) Then
                FindSharedDescriptionRow = r
                Exit Function
            End If

        Next c

    Next r

    FindSharedDescriptionRow = 0

End Function

Private Function CountPopulatedBlocks( _
    ByVal ws As Worksheet, _
    ByRef blocks() As CellFieldBlock, _
    ByVal blockCount As Long) As Long

    Dim i As Long
    Dim n As Long

    For i = 1 To blockCount
        If BlockHasRelationPairs(ws, blocks(i)) Then
            n = n + 1
        End If
    Next i

    CountPopulatedBlocks = n

End Function

Private Function BlockHasRelationPairs( _
    ByVal ws As Worksheet, _
    ByRef block As CellFieldBlock) As Boolean

    Dim c As Long
    Dim lastDataCol As Long
    Dim iobValue As String
    Dim fieldName As String

    lastDataCol = LastUsedColumnForBlock( _
        ws:=ws, _
        iobRow:=block.iobRow, _
        fieldRow:=block.fieldRow, _
        descriptionRow:=0)

    For c = block.labelCol + 1 To lastDataCol

        iobValue = CellText(ws.cells(block.iobRow, c).Value2)
        fieldName = CellText(ws.cells(block.fieldRow, c).Value2)

        If Len(iobValue) > 0 And Len(fieldName) > 0 Then
            BlockHasRelationPairs = True
            Exit Function
        End If

    Next c

    BlockHasRelationPairs = False

End Function

Private Function LastUsedColumnForBlock( _
    ByVal ws As Worksheet, _
    ByVal iobRow As Long, _
    ByVal fieldRow As Long, _
    ByVal descriptionRow As Long) As Long

    Dim lastCol As Long
    Dim candidate As Long

    lastCol = ws.cells(iobRow, ws.Columns.Count).End(xlToLeft).Column

    candidate = ws.cells(fieldRow, ws.Columns.Count).End(xlToLeft).Column
    If candidate > lastCol Then lastCol = candidate

    If descriptionRow > 0 Then
        candidate = ws.cells(descriptionRow, ws.Columns.Count).End(xlToLeft).Column
        If candidate > lastCol Then lastCol = candidate
    End If

    LastUsedColumnForBlock = lastCol

End Function

Private Function ResolveRunNameOrBlank( _
    ByVal explicitRunName As String, _
    ByVal populatedBlockCountOnSheet As Long) As String

    explicitRunName = CleanText(explicitRunName)

    If populatedBlockCountOnSheet = 1 Then
        ResolveRunNameOrBlank = V1_SINGLE_RUN_NAME
        Exit Function
    End If

    If Len(explicitRunName) > 0 Then
        ResolveRunNameOrBlank = explicitRunName
        Exit Function
    End If

    ResolveRunNameOrBlank = vbNullString

End Function

Private Function ExtractExplicitRunName( _
    ByVal iobLabel As String, _
    ByVal fieldLabel As String) As String

    Dim s As String

    s = CleanText(iobLabel)

    s = Replace(s, "I/O/B", "", 1, -1, vbTextCompare)
    s = Replace(s, "IOB", "", 1, -1, vbTextCompare)
    s = CleanText(s)

    If Len(s) = 0 Then

        s = CleanText(fieldLabel)

        s = Replace(s, "Field name", "", 1, -1, vbTextCompare)
        s = Replace(s, "Field Name", "", 1, -1, vbTextCompare)
        s = Replace(s, "FIELD_NAME", "", 1, -1, vbTextCompare)
        s = Replace(s, "FIELDNAME", "", 1, -1, vbTextCompare)
        s = CleanText(s)

    End If

    ExtractExplicitRunName = s

End Function

Private Function BuildDescription( _
    ByVal descriptionBase As String, _
    ByVal fieldName As String, _
    ByVal runName As String) As String

    Dim result As String

    result = CleanText(descriptionBase)

    If Len(result) = 0 Then
        result = CleanText(DDLookupDescription(fieldName))
    End If

    If Len(result) = 0 Then
        result = CleanText(fieldName)
    End If

    If V1_APPEND_RUN_TO_DESCRIPTION Then
        If Len(CleanText(runName)) > 0 Then
            result = CleanText(result & " at " & runName)
        End If
    End If

    BuildDescription = CleanText(result)

End Function

Private Sub WriteCellFieldRelationHeaders(ByVal reportWs As Worksheet)

    With reportWs
        .cells(1, 1).value = "SOURCE_TAB"
        .cells(1, 2).value = "CELL_ADDRESS"
        .cells(1, 3).value = "RUN"
        .cells(1, 4).value = "FIELD_NAME_AT_RUN"
        .cells(1, 5).value = "I/O/B"
        .cells(1, 6).value = "DESCRIPTION"
    End With

End Sub

Private Function Row2Address( _
    ByVal ws As Worksheet, _
    ByVal sourceColumn As Long) As String

    Row2Address = ws.cells(2, sourceColumn).Address( _
        RowAbsolute:=False, _
        ColumnAbsolute:=False)

End Function

Private Function CellText(ByVal value As Variant) As String

    If IsError(value) Then
        CellText = vbNullString
    Else
        CellText = CleanText(CStr(value))
    End If

End Function

Private Function CleanText(ByVal textValue As String) As String

    Dim s As String

    s = CStr(textValue)
    s = Replace(s, vbCr, " ")
    s = Replace(s, vbLf, " ")
    s = Replace(s, vbTab, " ")
    s = Trim$(s)

    Do While InStr(1, s, "  ", vbBinaryCompare) > 0
        s = Replace(s, "  ", " ")
    Loop

    CleanText = s

End Function

Private Function IsIOBLabel(ByVal labelText As String) As Boolean

    Dim s As String

    s = UCase$(CleanText(labelText))

    IsIOBLabel = _
        InStr(1, s, "I/O/B", vbTextCompare) > 0 Or _
        InStr(1, s, "IOB", vbTextCompare) > 0

End Function

Private Function IsFieldNameLabel(ByVal labelText As String) As Boolean

    Dim s As String

    s = UCase$(CleanText(labelText))

    IsFieldNameLabel = _
        InStr(1, s, "FIELD NAME", vbTextCompare) > 0 Or _
        InStr(1, s, "FIELD_NAME", vbTextCompare) > 0 Or _
        InStr(1, s, "FIELDNAME", vbTextCompare) > 0

End Function

Private Function IsDescriptionLabel(ByVal labelText As String) As Boolean

    Dim s As String

    s = UCase$(CleanText(labelText))

    IsDescriptionLabel = _
        s = "DESCRIPTION" Or _
        s = "FIELD DESCRIPTION" Or _
        s = "FIELD_DESCRIPTION" Or _
        s = "DESC" Or _
        s = "DESCRIPTIONS" Or _
        InStr(1, s, "DESCRIPTION", vbTextCompare) > 0

End Function

Private Function IsGeneratedV1ReportSheet(ByVal sheetName As String) As Boolean

    Select Case UCase$(CleanText(sheetName))

        Case UCase$(V1_CELL_FIELD_RELATION_SHEET), _
             "V1_ROW2_FORMULAS", _
             "V1_NAMED_RANGES"

            IsGeneratedV1ReportSheet = True

        Case Else

            IsGeneratedV1ReportSheet = False

    End Select

End Function

Private Sub DeleteWorksheetIfExists( _
    ByVal wb As Workbook, _
    ByVal sheetName As String)

    Dim ws As Worksheet

    For Each ws In wb.Worksheets

        If StrComp(ws.Name, sheetName, vbTextCompare) = 0 Then
            ws.Delete
            Exit Sub
        End If

    Next ws

End Sub

Private Function DDLookupDescription(ByVal fieldName As String) As String

    Static dd As Object
    Dim key As String

    If dd Is Nothing Then
        Set dd = CreateObject("Scripting.Dictionary")
        dd.CompareMode = vbTextCompare
        LoadEmbeddedDDDescriptions dd
    End If

    key = DDKey(fieldName)

    If dd.Exists(key) Then
        DDLookupDescription = CleanText(CStr(dd(key)))
    Else
        DDLookupDescription = vbNullString
    End If

End Function

Private Function DDKey(ByVal fieldName As String) As String

    Dim s As String
    Dim p As Long

    s = CleanText(fieldName)

    If Len(s) = 0 Then
        DDKey = vbNullString
        Exit Function
    End If

    p = InStrRev(s, ".")
    If p > 0 And p < Len(s) Then
        s = Mid$(s, p + 1)
    End If

    DDKey = UCase$(CleanText(s))

End Function

Private Sub AddDD( _
    ByVal dd As Object, _
    ByVal fieldName As String, _
    ByVal descriptionText As String)

    Dim key As String

    key = DDKey(fieldName)

    If Len(key) > 0 Then
        dd(key) = CleanText(descriptionText)
    End If

End Sub
